```plantuml
@startuml
title Проход на стадион по QR-коду через турникет

actor "User" as User
participant "Football Ticket System" as FTS
database "Pass Storage" as PassStorage
actor "Security Service" as Security

FTS -> User : requestScan()
User -> FTS : scanTicket()
activate FTS

FTS -> PassStorage : getTicketInfo(scannedData)
activate PassStorage
PassStorage -> PassStorage : checkAccess(scannedData, currentEvent)
PassStorage --> FTS : ticketInfo
deactivate PassStorage

alt Access denied
    FTS -> User : alert(reason)
    FTS -> Security : notifyAccessDenied(ticketInfo, reason)
    FTS -> FTS : waitForUserAction(10s)
else Access allowed
    FTS -> User : showSeatInfo(sector, row, seat)
    FTS -> FTS : unlockTurnstile()
    User -> FTS : passThrough()
    FTS -> FTS : lockTurnstile()

    FTS -> PassStorage : registerEntry(scannedData)
    activate PassStorage
    PassStorage -> PassStorage : markTicketUsed(scannedData, currentEvent)
    PassStorage --> FTS : entryRegistered
    deactivate PassStorage

    FTS -> Security : sendOccupiedSeat(sector, row, seat)
end

FTS -> FTS : toWaitingMode()
deactivate FTS

@enduml
```
