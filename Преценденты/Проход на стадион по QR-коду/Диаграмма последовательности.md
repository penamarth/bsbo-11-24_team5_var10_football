```plantuml
@startuml
title Проход на стадион по QR-коду через FTS

actor "User" as User
participant "FTS" as FTS
participant "EventRepository" as EventRepository
participant "PassRepository" as PassRepository
participant "Pass" as Pass
participant "ITurnstileAdapter" as Adapter
participant "ExternalTurnstileAPI" as External
participant "ConsolePanes" as Panes

User -> FTS : startUseTicket()
activate FTS

loop Каждый вызванный метод
    FTS -> Panes : log(line)
end

FTS -> FTS : selectEntryMatch()
FTS -> EventRepository : getEvents()
activate EventRepository
EventRepository --> FTS : eventsList
deactivate EventRepository
FTS --> User : displayMatches(matchesList)
User -> FTS : matchId
FTS -> EventRepository : findMatchById(matchId)
activate EventRepository
EventRepository --> FTS : match
deactivate EventRepository

FTS -> User : requestScan()
User -> FTS : passId

FTS -> PassRepository : getPassById(passId)
activate PassRepository
PassRepository --> FTS : pass
deactivate PassRepository

FTS -> FTS : проверить isRefunded
FTS -> Pass : checkAccess(matchId)
activate Pass
Pass --> FTS : accessAllowed
deactivate Pass

alt Access denied
    FTS --> User : alert(reason)
else Access allowed
    FTS --> User : showSeatInfo(seatId)
    User -> FTS : turnstileId

    FTS -> FTS : unlockTurnstile(turnstileId)
    FTS -> Adapter : unlock(turnstileId)
    activate Adapter
    Adapter -> External : sendOpenCommand(deviceId, token)
    activate External
    External --> Adapter : true
    deactivate External
    Adapter --> FTS : true
    deactivate Adapter

    FTS -> Pass : markUsed(matchId)
    activate Pass
    Pass --> FTS : true
    deactivate Pass

    FTS -> FTS : lockTurnstile(turnstileId)
    FTS -> Adapter : lock(turnstileId)
    activate Adapter
    Adapter -> External : sendCloseCommand(deviceId, token)
    activate External
    External --> Adapter : true
    deactivate External
    Adapter --> FTS : true
    deactivate Adapter
end

deactivate FTS
@enduml
```
