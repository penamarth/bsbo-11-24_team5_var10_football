```plantuml
@startuml
title Покупка билетов и абонементов через КСО

actor "User" as User
participant "Football Ticket System" as FTS
database "Pass Storage" as PassStorage
participant "Email Notification System" as EmailSystem

User -> FTS : startPurchase()
activate FTS
FTS -> PassStorage : getAvailableEvents()
activate PassStorage
PassStorage --> FTS : eventsList
deactivate PassStorage
FTS -> User : showEvents(eventsList)

User -> FTS : selectEvent(eventId)
FTS -> PassStorage : getSeatMap(eventId)
activate PassStorage
PassStorage --> FTS : seatMap
deactivate PassStorage
FTS -> User : showSeatMap(seatMap)

User -> FTS : selectSeat(seatId)
FTS -> PassStorage : reserveSeat(eventId, seatId)
activate PassStorage
PassStorage -> PassStorage : checkSeatAvailability(eventId, seatId)
PassStorage --> FTS : reservation
deactivate PassStorage

alt Seat taken
    FTS -> User : alert("Seat is already taken")
    FTS -> FTS : waitForUserAction(10s)
else Seat reserved
    FTS -> User : showPurchaseDetails(price)
    User -> FTS : confirmPurchase(paymentType)
    FTS -> User : requestEmail()
    User -> FTS : enterEmail(email)
    FTS -> User : requestPayment(price)
    User -> FTS : makePayment(amount)
    FTS -> User : dispenseChange(change)

    FTS -> PassStorage : createTicket(reservation)
    activate PassStorage
    PassStorage -> PassStorage : markSeatOccupied(eventId, seatId)
    PassStorage --> FTS : ticket
    deactivate PassStorage

    FTS -> User : printTicket(ticket)
    FTS -> User : showReceipt()
    FTS -> User : printReceipt()

    FTS -> EmailSystem : sendTicketEmail(email, ticket)
    activate EmailSystem
    EmailSystem --> FTS : emailSent
    deactivate EmailSystem
end

FTS -> FTS : toMainMenu()

@enduml
```
