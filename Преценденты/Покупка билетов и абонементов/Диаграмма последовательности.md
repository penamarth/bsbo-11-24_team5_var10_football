```plantuml 
@startuml
title Покупка билетов и абонементов через КСО

actor "User" as User
participant "FTS" as FTS
participant "EventStorage" as EventStorage
participant "Event" as Event
participant "IBankAdapter" as Bank
participant "SberbankAPI" as Sber
participant "PassStorage" as PassStorage

User -> FTS : getAvailableEvents()
activate FTS
FTS -> EventStorage : loadEvents()
activate EventStorage
EventStorage --> FTS : eventsList
deactivate EventStorage
FTS --> User : eventsList

User -> FTS : getSeatMap(eventId)
FTS -> EventStorage : findEventById(eventId)
activate EventStorage
EventStorage --> FTS : event
deactivate EventStorage
FTS -> Event : findMatchById(matchId)
activate Event
Event --> FTS : match
deactivate Event
FTS --> User : seatMap

User -> FTS : reserveSeat(matchId, seatId)
FTS -> EventStorage : reserveSeat(matchId, seatId, timeout)
activate EventStorage
EventStorage --> FTS : reservation
deactivate EventStorage

alt Seat taken
    FTS --> User : seatTaken
else Seat reserved
    User -> FTS : createPurchase()

    alt PaymentTypeCard
        FTS -> Bank : startPurchase()
        activate Bank
        Bank -> Sber : registerOrder()
        activate Sber
        Sber --> Bank : orderStatus
        deactivate Sber
        Bank --> FTS : paymentResult
        deactivate Bank
    else PaymentTypeCash
        note over FTS : оплата наличными,\nбанк не вызывается
    end

    alt Payment declined
        FTS --> User : paymentDeclined
    else Payment approved
        create participant "Ticket" as Ticket
        FTS -> Ticket : <<create>>
        note right of Ticket : для абонемента —\nSubscription
        FTS -> PassStorage : addPass(ticket)
        FTS -> Event : setSeatStatus(matchId, seatId, SeatStatusSold)
        FTS -> FTS : sendTicketEmail(email, passId)
        FTS --> User : true
    end
end

deactivate FTS
@enduml
```
