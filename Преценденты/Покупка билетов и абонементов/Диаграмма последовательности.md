```plantuml
@startuml
title Покупка билета и абонемента через FTS

actor "User" as User
participant "FTS" as FTS
participant "EventRepository" as EventRepository
participant "PurchaseStrategy" as Strategy
participant "IBankAdapter" as Bank
participant "SberbankAPI" as Sber
participant "PassRepository" as PassRepository
participant "ConsolePanes" as Panes

User -> FTS : startPurchase()
activate FTS

loop Каждый вызванный метод
    FTS -> Panes : log(line)
end

FTS -> FTS : getAvailableEvents()
FTS -> EventRepository : getEvents()
activate EventRepository
EventRepository --> FTS : eventsList
deactivate EventRepository
FTS --> User : selectEvent(eventsList)
User -> FTS : eventId

FTS -> FTS : requestPurchaseType()
User -> FTS : purchaseType

FTS -> Strategy : needsMatchSelection()
Strategy --> FTS : true/false

alt Билет
    FTS -> FTS : selectMatch(event)
    User -> FTS : matchId
    FTS -> Strategy : getMatches(event, match)
else Абонемент
    FTS -> Strategy : getMatches(event, null)
end
Strategy --> FTS : matches

FTS -> FTS : getSeatMap(matchId)
FTS -> EventRepository : findMatchById(matchId)
activate EventRepository
EventRepository --> FTS : match
deactivate EventRepository
FTS -> FTS : match.stadiumMap.clone()
FTS -> Strategy : getTitle(event, match)
FTS --> User : selectSeat(title, map, match)
User -> FTS : seatId

FTS -> FTS : reserveSeats(matches, seatId)
FTS -> EventRepository : reserveSeat(matchId, seatId, timeout)
activate EventRepository
EventRepository --> FTS : reservation
deactivate EventRepository

alt seatTaken
    FTS --> User : seatTaken
else seatReserved
    FTS -> Strategy : calculatePrice(seat, matches)
    Strategy --> FTS : price
    FTS --> User : showPurchaseDetails(event, match, seat, price)
    User -> FTS : confirm
    FTS --> User : requestEmail()
    User -> FTS : email
    FTS -> Strategy : getDescription(event, match, seat)

    FTS -> Bank : startPurchase(description, price)
    activate Bank
    Bank -> Sber : createPurchase(token, description, amount)
    activate Sber
    note over Sber : покупатель вводит данные карты\nна странице банка и подтверждает\nплатёж (3-D Secure)
    Sber --> Bank : url
    deactivate Sber
    Bank --> FTS : url
    deactivate Bank

    FTS -> Bank : confirmPurchase(url)
    activate Bank
    Bank -> Sber : purchaseStatus(url)
    activate Sber
    Sber --> Bank : "success"
    deactivate Sber
    Bank --> FTS : "success"
    deactivate Bank

    FTS -> Strategy : createPass(event, matches, seat, price)
    activate Strategy
    Strategy --> FTS : pass
    deactivate Strategy

    FTS -> PassRepository : createPass(pass)
    activate PassRepository
    PassRepository --> FTS : pass.id
    deactivate PassRepository

    loop Для каждого матча из matches
        FTS -> EventRepository : findMatchById(matchId)
        FTS -> FTS : event.setSeatStatus(matchId, seatId, Sold)
    end

    FTS -> FTS : showPurchaseConfirmation(pass)
    FTS -> FTS : sendTicketEmail(pass)
    FTS --> User : true
end

deactivate FTS
@enduml
```
