```
@startuml
actor "Менеджер" as Manager

participant "FTS" as Controller
participant "Event Storage" as DB
participant "Event" as Event
participant "Match" as Match


Manager -> Controller : getAvailableEvents()
activate Controller
Controller -> DB : getEvents()
activate DB
DB --> Controller : events

Manager -> Controller : createNewEvent(eventName, startTimestamp, endTimestamp)
Controller -> DB : createEvent(eventName, startTimestamp, endTimestamp)
DB -> Event : Event(eventName, startTimestamp, endTimestamp)
activate Event
Event --> DB : event
deactivate Event
DB --> Controller : event
Manager -> Controller : createNewMatch(matchName, priceCoefficient, startTimestamp, endTimestamp)
Controller -> DB : createMatch(matchName, priceCoefficient, startTimestamp, endTimestamp)

activate Match
DB -> Match : createMatch(eventId, match)
Match --> DB : bool
deactivate Match
DB -> Event : addMatch(match)
activate Event
Event --> DB : bool
deactivate Event
DB --> Controller : successStatus

deactivate Event
Controller -> Controller : addEvent()
Controller --> Manager : displayMap(match)

loop Для каждого изменяемого места
    Manager -> Controller : updateSeat(matchId, seatId, status)
    Controller -> Event : setSeatStatus(matchId, seatId, status)
    activate Event
    Event -> Event : findMatchById(id)
    Event -> Match : setSeatStatus(matchId, seatId, status)
    Match --> Event : successStatus
    deactivate Match
    Event --> Controller : successStatus
    deactivate Event
    Controller --> Manager : displayStatus()
end

Manager -> Controller : saveEvent()
Controller -> DB : updateEvent(events)
DB --> Controller : successStatus
deactivate DB
Controller --> Manager : displayConformation()
deactivate Controller
@enduml
```
