@startuml

@startuml

actor "Менеджер" as Manager

participant "FTS" as Controller
participant "Event" as Event
participant "Match" as Match
participant "StadiumMap" as Map
database "Event Storage" as DB

Manager -> Controller : createEvent()
activate Controller
Controller -> DB : loadEvents()
activate DB
DB --> Controller : eventsList
deactivate DB
Manager -> Controller : createNew(eventName)
Controller -> Event : createNewEvent(eventName)
activate Event
Event --> Manager : askForMatchName()
Manager -> Controller : matchName
Controller -> Event : createNewMatch(matchName)
Event -> Match : createNewMatch(matchName)
activate Match
Match --> Event : match
Match -> Map : createDefaultMap()
activate Map
Map --> Match : map
deactivate Map
Match --> Event : successStatus
deactivate Match
Event --> Controller : successStatus
Controller -> Controller : appendToList(eventsList, event)
Controller --> Manager : displayMap(event)

loop Для каждого изменяемого места
    Manager -> Controller : updateSeat(matchName, seatId, status)
    Controller -> Event : setNewSeatStatus(matchName, seatId, status)
    activate Event
    Event -> Event : findMatch(matchName)
    Event -> Match : setNewSeatStatus(seatId, status)
    activate Match
    Match -> Map : setStatus(seatId, status)
    activate Map
    Map --> Match : successStatus
    deactivate Map
    Match --> Event : successStatus
    deactivate Match
    Event --> Controller : successStatus
    deactivate Event
    Controller --> Manager : displayNewStatus()
end

Manager -> Controller : saveEvent()
Controller -> DB : save(event)
activate DB
DB --> Controller : successStatus
deactivate DB
Controller --> Manager : displayConformation()
deactivate Controller

@enduml

@enduml
