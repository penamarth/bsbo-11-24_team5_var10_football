```plantuml
@startuml
title Возврат билета и абонемента через FTS

actor "User" as User
participant "FTS" as FTS
participant "PassRepository" as PassRepository
participant "RefundStrategy" as Strategy
participant "IBankAdapter" as Bank
participant "SberbankAPI" as Sber
participant "EventRepository" as EventRepository
participant "ConsolePanes" as Panes

User -> FTS : startRefund()
activate FTS

loop Каждый вызванный метод
    FTS -> Panes : log(line)
end

FTS -> User : requestScan()
User -> FTS : passId

FTS -> PassRepository : checkRefundEligibility(passId)
activate PassRepository
PassRepository --> FTS : refundAllowed
deactivate PassRepository

alt Возврат запрещён
    FTS --> User : scanAnotherPass()
else Возврат разрешён
    FTS -> PassRepository : getPassById(passId)
    activate PassRepository
    PassRepository --> FTS : pass
    deactivate PassRepository

    FTS --> User : showRefundDetails(pass)
    User -> FTS : confirm

    alt Оплата картой
        FTS -> FTS : refundToCard(pass)
        FTS -> Bank : refund(credentials, description, price)
        activate Bank
        Bank -> Sber : transfer(token, receiver, description, amount)
        activate Sber
        Sber --> Bank : true
        deactivate Sber
        Bank --> FTS : true
        deactivate Bank
    else Оплата наличными
        FTS -> FTS : refundWithCash(pass)
    end

    FTS -> PassRepository : refund(passId)
    activate PassRepository
    PassRepository --> FTS : true
    deactivate PassRepository

    FTS -> Strategy : releaseSeats(pass, eventRepo)
    activate Strategy
    alt Тип пропуска Ticket
        Strategy -> EventRepository : refundSeat(matchId, seatId)
    else Тип пропуска Subscription
        loop Для каждого matchId
            Strategy -> EventRepository : refundSeat(matchId, seatId)
        end
    end
    deactivate Strategy

    FTS -> FTS : giveReceipt(pass)
    FTS -> FTS : sendEmail(email, text)
    FTS -> FTS : printReceipt(text)
    FTS --> User : waitForEnter()
end

deactivate FTS
@enduml
```
