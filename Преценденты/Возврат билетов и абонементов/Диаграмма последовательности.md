```plantuml
@startuml
title Возврат билетов и абонементов через КСО

actor "User" as User
participant "Football Ticket System" as FTS
participant "Pass Storage" as PassStorage
participant "Bank Adapter" as Bank
participant "Audit Storage" as Audit

User -> FTS : startRefund()
activate FTS

FTS -> User : requestScan()
deactivate FTS

User -> FTS : scanPass()
activate FTS

FTS -> PassStorage : checkRefundEligibility(passId)
activate PassStorage
deactivate FTS
PassStorage --> FTS : refundAllowed
deactivate PassStorage
activate FTS

group not refundable
	FTS -> User : askRescan()
	deactivate FTS
	User -> FTS : answer
	activate FTS
end

FTS -> PassStorage : getPassById(passId)
activate PassStorage
deactivate FTS
PassStorage --> FTS : pass
deactivate PassStorage
activate FTS

FTS -> User : showRefundDetails(pass)
deactivate FTS

User -> FTS : confirmRefund()
activate FTS

group payment by card
	FTS -> Bank : refund(credentials, description, amount)
	activate Bank
	deactivate FTS
	Bank --> FTS : refunded
	deactivate Bank
	activate FTS

	group card refund failed
		FTS -> User : offerCashRefund()
		deactivate FTS
		User -> FTS : answer
		activate FTS
	end
end

group payment by cash
	FTS -> FTS : refundWithCash()
end

FTS -> PassStorage : refund(passId)
activate PassStorage
deactivate FTS
PassStorage --> FTS : refunded
deactivate PassStorage
activate FTS

FTS -> User : giveReceipt()
FTS -> Audit : saveToAudit(text)
activate Audit
deactivate Audit
FTS -> FTS : toMainMenu()

deactivate FTS
@enduml
```
