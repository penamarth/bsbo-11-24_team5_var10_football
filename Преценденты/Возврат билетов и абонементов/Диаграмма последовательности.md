![Диаграмма](<../../source/Диаграмма последовательности - Возврат покупок и абонементов.png>)

```plantuml
@startuml
title Возврат билетов и абонементов через КСО

actor "User" as User
participant "Football Ticket System" as FTS
participant "Pass Storage" as PassStorage

User -> FTS : startRefund()
activate FTS

FTS -> User : requestScan()
deactivate FTS

User -> FTS : scanPass()
activate FTS

FTS -> PassStorage : checkRefundEligibility(scannedData)
activate PassStorage
deactivate FTS
PassStorage --> FTS : refundAllowed
deactivate PassStorage
activate FTS

FTS -> User : showRefundDetails(refundInfo)
deactivate FTS

User -> FTS : confirmRefund()
activate FTS

FTS -> PassStorage : refund(scannedData)
activate PassStorage
deactivate FTS

PassStorage --> FTS : refundCreated
deactivate PassStorage
activate FTS

FTS -> User : giveReceipt()

FTS -> FTS : saveToAudit(text)

FTS -> FTS : toMainMenu()

@enduml
```
