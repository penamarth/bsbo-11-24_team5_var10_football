```plantuml
@startuml
title Возврат билетов и абонементов через КСО

actor "User" as User
participant "Football Ticket System" as FTS
database "Ticket Storage" as TicketStorage

User -> FTS : startRefund()
activate FTS
FTS -> User : requestScan()
User -> FTS : scanTicket()

FTS -> TicketStorage : getTicketInfo(scannedData)
activate TicketStorage
TicketStorage -> TicketStorage : checkRefundEligibility(scannedData)
TicketStorage --> FTS : ticketInfo
deactivate TicketStorage

alt Refund denied
    FTS -> User : alert("Call staff")
    FTS -> FTS : waitForUserAction(10s)
else Refund allowed
    FTS -> User : showRefundDetails(amount)
    User -> FTS : confirmRefund()
    
    FTS -> TicketStorage : createRefund(scannedData)
    activate TicketStorage
    TicketStorage -> TicketStorage : invalidateTicket(scannedData)
    TicketStorage --> FTS : refundCreated
    deactivate TicketStorage

    FTS -> User : dispenseCash(amount)
    FTS -> User : showReceipt()
    FTS -> User : printReceipt()
end

FTS -> FTS : toMainMenu()

@enduml
```
