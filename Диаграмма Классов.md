@startuml
scale 0.5
!define BOUNDED_CONTEXT(name) rectangle name <<Ограниченный контекст>>
!define AGGREGATE(name) rectangle name <<Агрегат>>

skinparam <<Ограниченный контекст>> {
    roundcorner 100
    PackageTitleAlignment 60

BOUNDED_CONTEXT("Events") as EventsContext {
    interface IPrototype<T> <<Интерфейс>> {
        + clone() : T
    }
    interface EventComponent <<Интерфейс>> {
        + getId() : int
        + getName() : string
        + getStartTimestamp() : int
        + getEndTimestamp() : int
    }

    AGGREGATE("Событие") {
        class Event <<Корень агрегата>> <<Композит>> {
            - id: int
            - name: string
            - startTimestamp: int
            - endTimestamp: int
            - children: List<EventComponent>
            + addChild(c: EventComponent)
            + removeChild(id: int) : bool
            + getChildren() : List<EventComponent>
            + findMatchById(matchId: int) : Match
            + setSeatStatus(matchId: int, seatId: int, status: SeatStatus) : bool
        }
    }

    AGGREGATE("Матч") {
        class Match <<Корень агрегата>> <<Лист>> {
            - id: int
            - name: string
            - priceCoefficient: double
            - startTimestamp: int
            - endTimestamp: int
            - seatStatuses: Map<int, SeatStatus>
            - stadiumMap: StadiumMap
            + setSeatStatus(seatId: int, status: SeatStatus) : bool
        }
        class StadiumMap <<Объект-Значение>> <<Прототип>> {
            - id: int
            - name: string
            - layoutImage: string
            - seats: List<Seat>
            - turnstiles: List<int>
            + clone() : StadiumMap
        }
        class Seat <<Сущность>> {
            - id: int
            - sector: int
            - row: int
            - column: int
            - price: double
        }
        class SeatStatus <<Перечисление>> {
            Free
            Booked
            Sold
            Blocked
        }

        Match --> StadiumMap
        StadiumMap *-- Seat
        Match --> SeatStatus
    }

    class EventRepository <<Репозиторий>> {
        - events: List<Event>
        - stadiumMaps: List<StadiumMap>
        + getEvents() : List<Event>
        + findEventById(id: int) : Event
        + findMatchById(id: int) : Match
        + getStadiumMaps() : List<StadiumMap>
        + findStadiumMapById(id: int) : StadiumMap
        + createStadiumMap(map: StadiumMap)
        + createEvent(e: Event)
        + createMatch(eventId: int, match: Match) : bool
        + reserveSeat(matchId: int, seatId: int, timeoutSeconds: int) : bool
        + checkBooking(matchId: int, seatId: int) : bool
        + releaseSeat(matchId: int, seatId: int) : bool
        + refundSeat(matchId: int, seatId: int) : bool
        + updateEvent(id: int, newEvent: Event) : bool
        + updateMatch(id: int, newMatch: Match) : bool
    }

    IPrototype <|.. StadiumMap
    EventComponent <|.. Event
    EventComponent <|.. Match
    Event o-- "*" EventComponent
    EventRepository o-- "*" Event
    EventRepository o-- "*" Match
    EventRepository o-- "*" StadiumMap
}

BOUNDED_CONTEXT("Passes") as PassesContext {
    AGGREGATE("Пропуск") {
        abstract class Pass <<Корень агрегата>> {
            - id: int
            - seatId: int
            - price: double
            - email: string
            - isRefunded: bool
            - paymentMethod: PaymentType
            - paymentCredentials: string
            - QRCode: string
            ~ UsageChanged: event Action<Pass>
            + checkAccess(matchId: int) : bool
            + markUsed(matchId: int) : bool
            + checkRefundEligibility() : bool
            # notifyUsageChanged()
        }
        class Ticket <<Сущность>> {
            - matchId: int
            - used: bool
        }
        class Subscription <<Сущность>> {
            - matchIds: Set<int>
            - usedMatches: Set<int>
            - startTimestamp: int
            - endTimestamp: int
        }
        class PaymentType <<Перечисление>> {
            Card
            Cash
        }

        Pass <|-- Ticket
        Pass <|-- Subscription
        Pass --> PaymentType
    }

    class PassRepository <<Репозиторий>> {
        - passes: List<Pass>
        + PassChanged: event Action<Pass>
        + createPass(pass: Pass)
        + checkRefundEligibility(passId: int) : bool
        + refund(passId: int) : bool
        + getPassById(passId: int) : Pass
        + getPassesWithFilter(f: Func<Pass, bool>) : List<Pass>
        + coversAnyMatch(pass: Pass, matchIds: Set<int>) : bool
        - notifyPassChanged(pass: Pass)
    }

    PassRepository o-- "*" Pass
    PassRepository ..> Pass : подписка на UsageChanged

    interface PurchaseStrategy <<Интерфейс>> <<Стратегия>> {
        + needsMatchSelection() : bool
        + getMatches(ev: Event, selectedMatch: Match) : List<Match>
        + getTitle(ev: Event, match: Match) : string
        + calculatePrice(seat: Seat, matches: List<Match>) : double
        + getDescription(ev: Event, match: Match, seat: Seat) : string
        + createPass(ev: Event, matches: List<Match>, seat: Seat, price: double) : Pass
    }
    class TicketPurchaseStrategy <<Стратегия>>
    class SubscriptionPurchaseStrategy <<Стратегия>>

    interface RefundStrategy <<Интерфейс>> <<Стратегия>> {
        + releaseSeats(pass: Pass, eventRepo: EventRepository)
    }
    class TicketRefundStrategy <<Стратегия>>
    class SubscriptionRefundStrategy <<Стратегия>>

    PurchaseStrategy <|.. TicketPurchaseStrategy
    PurchaseStrategy <|.. SubscriptionPurchaseStrategy
    RefundStrategy <|.. TicketRefundStrategy
    RefundStrategy <|.. SubscriptionRefundStrategy
    PurchaseStrategy ..> Pass
    RefundStrategy ..> Pass
}

BOUNDED_CONTEXT("Analytics") as AnalyticsContext {
    interface AnalyticsSubject <<Интерфейс>> <<Subject>> {
        + attach(o: AnalyticsObserver)
        + detach(o: AnalyticsObserver)
        + notifyObservers(eventId: int)
    }
    interface AnalyticsObserver <<Интерфейс>> <<Observer>> {
        + update(eventId: int, ticketsSold: int, revenue: double)
    }

    AGGREGATE("Отчёт") {
        class Report <<Корень агрегата>> <<Concrete Observer>> {
            - eventId: int
            - totalRevenue: double
            - ticketsSold: int
            - attendanceCount: int
            - updatedAt: int
            - passes: List<Pass>
            + update(eventId: int, ticketsSold: int, revenue: double)
            + calculateMetrics()
            + toCsv() : string
        }
    }

    AnalyticsObserver <|.. Report
}
