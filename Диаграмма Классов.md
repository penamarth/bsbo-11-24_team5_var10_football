```plantuml
@startuml
rectangle "Analytics" {
	class AuditStorage {
		entries: list<string>
		+ save(text: string): void
		+ getAll(): list<string>
	}

	class Report {
		eventId: int
		totalRevenue: double
		ticketsSold: int
		attendanceCount: int
		updatedAt: int
		+ calculateMetrics(): void
	}
}

rectangle "Events" {
	class EventStorage {
		events: list<Event>
		+ getEvents(): list<Event>
		+ findEventById(id: int): Event
		+ findMatchById(id: int): Match
		+ createEvent(e: Event): void
		+ createMatch(eventId: int, match: Match): bool
		+ reserveSeat(matchId: int, seatId: int): bool
		+ updateEvent(id: int, newEvent: Event): bool
		+ updateMatch(id: int, newMatch: Match): bool
	}
	class Event {
		id: int
		name: string
		matches: list<Match>
		startTimestamp: int
		endTimestamp: int
		+ findMatchById(matchId: int): Match
		+ addMatch(match: Match): void
		+ setSeatStatus(matchId: int, seatId: int, status: SeatStatus): bool
	}

	class Match {
		id: int
		name: string
		priceCoefficient: double
		stadiumMap: StadiumMap
		startTimestamp: int
		endTimestamp: int
		seatStatuses: map<int, SeatStatus>
		+ setSeatStatus(seatId: int, status: SeatStatus): bool
	}

	class StadiumMap {
		name: string
		layoutImage: string
		seats: list<Seat>
		turnstiles: list<int>
	}

	class Seat {
		id: int
		sector: int
		row: int
		column: int
		price: double
	}

	enum SeatStatus {
		Free
		Booked
		Sold
		Blocked
	}
}

rectangle "Passes" {
	class PassStorage {
		passes: list<Pass>
		+ createPass(pass: Pass): void
		+ checkRefundEligibility(passId: int): bool
		+ refund(passId: int): bool
		+ getPassById(passId: int): Pass
	}

	abstract class Pass {
		id: int
		seatId: int
		price: double
		email: string
		isRefunded: bool
		paymentMethod: PaymentType
		paymentCredentials: string
		+ checkAccess(matchId: int): bool
		+ markUsed(matchId: int): bool
		+ checkRefundEligibility(): bool
	}

	class Ticket extends Pass {
		matchId: int
		used: bool
	}

	class Subscription extends Pass {
		matchIds: set<int>
		usedMatches: set<int>
		startTimestamp: int
		endTimestamp: int
	}


	enum PaymentType {
		Card
		Cash
	}
}


rectangle "Banking" {
	interface IBankAdapter {
		+ refund(receiver: string, description: string, amount: double): bool
		+ startPurchase(description: string, amount: double): string
		+ confirmPurchase(url: string): string
	}

	class SberbankAdapter {
		+ refund(receiver: string, description: string, amount: double): bool
		+ startPurchase(description: string, amount: double): string
		+ confirmPurchase(url: string): string
	}

	class SberbankAPI {
		+ transfer(token: string, receiver: string, description: string, amount: double): bool
		+ createPurchase(token: string, description: string, amount: double): string
		+ purchaseStatus(url: string): string
	}
}

rectangle "Application" {
	class FTS {
		passStorage: PassStorage
		eventStorage: EventStorage
		auditStorage: AuditStorage
		bank: IBankAdapter
		reports: list<Report>
		+ run(): void
		+ startRefund(): void
		+ requestScan(): int
		+ showRefundDetails(pass: Pass): bool
		+ refundToCard(pass: Pass): bool
		+ refundWithCash(pass: Pass): bool
		+ giveReceipt(pass: Pass): void
		+ saveToAudit(text: string): void
		+ getReport(eventId: int): Report
		+ getAvailableEvents(): list<Event>
		+ getSeatMap(matchId: int): StadiumMap
		+ unlockTurnstile(turnstileId: int): bool
		+ lockTurnstile(turnstileId: int): bool
		- sendEmail(email: string, text: string): bool
		- printReceipt(text: string): bool
	}
}

FTS --> EventStorage
FTS --> PassStorage
FTS --> AuditStorage
FTS --> IBankAdapter
FTS o-- Report

PassStorage *-- Pass
EventStorage *-- Event

Event *-- Match
Match --> StadiumMap
StadiumMap *-- Seat

Pass --> PaymentType
Match --> SeatStatus

SberbankAdapter ..|> IBankAdapter
SberbankAdapter ..> SberbankAPI
@enduml
```
