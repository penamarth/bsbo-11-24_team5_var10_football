@startuml
enum PaymentType {
	PaymentTypeCard
	PaymentTypeCash
}

enum SeatStatus {
	SeatStatusFree
	SeatStatusBooked
	SeatStatusSold
	SeatStatusBlocked
}

class FTS {
	passStorage: PassStorage
	eventStorage: EventStorage
	bankAPI: IBankAdapter
	reports: list<Report>
	+ requestScan(): void
	+ getReport(eventId: int): Report
	+ getAvailableEvents(): list<Event>
	+ getSeatMap(matchId: int): StadiumMap
	+ unlockTurnstile(stadiumId: int, turnstileId: int): bool
	+ lockTurnstile(stadiumId: int, turnstileId: int): bool
	+ createRefund(passId: int): bool
	+ createPurchase(): bool
    + showRefundDetails(refundInfo)
    + giveReceipt()
    + toMainMenu()
    + saveToAudit(text)
	- sendTicketEmail(email: string, passId: int): bool
}

class Report {
	eventId: int
	totalRevenue: double
	ticketsSold: int
	attendanceCount: int
	updatedAt: timestamp
	+ calculateMetrics()
}

class EventStorage {
	events: list<Event>
	+ loadEvents(): list<Event>
	+ findEventById(eventId: int): Event
	+ createEvent(event: Event): bool
	+ reserveSeat(matchId: int, seatId: int, timeout: timestamp)
	+ updateEvent(eventId: int, newEvent: Event)
	+ updateMatch(matchId: int, newMatch: Match)
}

class Event {
	id: int
	name: string
	matches: list<Match>
	start_timestamp: timestamp
	end_timestamp: timestamp
	+ findMatchById(matchId: int): Match
	+ createMatch(match: Match): bool
	+ setSeatStatus(matchId: int, seatId: int, status: SeatStatus)
}

class Match {
	id: int
	name: string
	stadiumMap: StadiumMap
	startTimestamp: timestamp
	endTimestamp: timestamp
	takenSits: list<SeatStatus>
}

class StadiumMap {
	name: string
	layoutImage: string
	seats: list<Seat>
	turnstiles: list<int>
}

class Seat {
	id: int
	sector: int
	row: int
	column: int
	price: double
}

class PassStorage {
	passes: list<Pass>
	+ addPass(pass: Pass): void
	+ checkRefundEligibility(passId: int): bool
	+ refund(passId: int): bool
}

interface IBankAdapter {
	+ refund(credentials: string): bool
	+ startPurchase(): string
}

class BankAdapter {
	+ refund(credentials: string): bool
	+ startPurchase(): string
	+ confirmPurcahse(url: string): bool
}

class SberbankAPI {
	+ refundTransaction()
	+ registerOrder()
}

class Pass {
	id: int
	seatId: int
	priceCoefficient: double
	email: string
	isRefunded: bool
	paymentMethod: PaymentType
	paymentCredentials: string
	+ checkAccess(matchId: int): bool
	+ markUsed(matchId: int): void
}

class Ticket extends Pass {
	matchId: int
	used: bool
}

class Subscription extends Pass {
	matchIds: list<int>
	used_matches: list<bool>
	start_timestamp: timestamp
	duration: timestamp
}

FTS --> EventStorage
FTS --> PassStorage
FTS --> IBankAdapter
FTS o-- Report

PassStorage *-- Pass

EventStorage *-- Event

Event *-- Match
Match --> StadiumMap : stadiumMap
StadiumMap *-- Seat

Pass --> PaymentType
Match --> SeatStatus

BankAdapter ..|> IBankAdapter
BankAdapter ..> SberbankAPI
@enduml
