```plantuml
@startuml
enum PaymentType {
	PaymentTypeCard
	PaymentTypeCash
}

enum SeatStatus {
	SeatStatusFree
	SeatStatusBooked
	SeatStatusSold
	SeatStatusBlocked
}

class FTS {
	passStorage: PassStorage
	eventStorage: EventStorage
    bankAPI: IBankAdapter
	reports: list<Report>
	+ requestScan(): void
    + getReport(eventId: int): Report
	+ getAvailableEvents(): list<Event>
	+ getSeatMap(eventId: int)
	+ unlockTurnstile(stadiumId: int, turnstileId: int): bool
	+ lockTurnstile(stadiumId: int, turnstileId: int): bool
    + createRefund(passId: int): bool
    + createPurchase(): bool
	- sendTicketEmail(email: string, passId: int): bool
}

class Report {
	eventId: int
	totalRevenue: double
	ticketsSold: int
	attendanceCount: int
	updatedAt: timestamp
	+ calculateMetrics()
}

class EventStorage {
	events: list<Event>
	matches: list<Match>
	+ loadEvents(): list<Event>
	+ findEventById(eventId: int): Event
	+ createEvent(event: Event): bool
	+ reserveSeat(matchId: int, seatId: int, timeout: timestamp)
	+ updateEvent(eventId: int, newEvent: Event)
	+ updateMatch(matchId: int, newMatch: Match)
}

class Event {
	id: int
	name: string
    matches: list<Match>
	start_timestamp: timestamp
	end_timestamp: timestamp
	+ findMatchById(matchId: int): Event
	+ createMatch(match: Match): bool
	+ setSeatStatus(matchId: int, seatId: int, status: SeatStatus)
}

class Match {
	id: int
    eventId: int
	name: string
	stadiumName: string
	startTimestamp: timestamp
	endTimestamp: timestamp
	takenSits: list<SeatStatus>
}

class StadiumMap {
	name: string
	layoutImage: string
	seats: list<Seat>
	turnstiles: list<int>
}

class Seat {
	id: int
	sector: int
	row: int
	column: int
	price: double
}

class PassStorage {
	passes: list<Pass>
	+ addPass(pass: Pass): void
	+ checkRefundEligibility(passId: int): bool
    + refundPass(passId: int): bool
}

interface IBankAdapter {
    + startRefund(credentials: string): bool
    + startPurchase(): string
}

class BankAdapter {
    + startRefund(credentials: string): bool
    + startPurchase(): string
}

class SberbankAPI {
    + refundTransaction()
    + registerOrder()
}

class Pass {
	id: int
	seatId: int
	priceCoefficient: double
	email: string
	isRefunded: bool
	paymentMethod: PaymentType
	paymentCredentials: string
	+ checkAccess(matchId: int): bool
	+ markUsed(matchId: int): void 
}

class Ticket extends Pass {
	match: string
	used: bool 
}

class Subscription extends Pass {
	matches: list<string>
	used_matches: list<bool>
	start_timestamp: timestamp
	duration: timestamp
}

FTS o-- EventStorage
FTS o-- PassStorage
FTS o-- IBankAdapter
FTS ..> Report

PassStorage *-- Pass
EventStorage *-- Event
EventStorage *-- Match

Event o-- Match
Match o-- StadiumMap

Seat o-- StadiumMap

PaymentType --> Pass
SeatStatus --> Seat

BankAdapter ..|> IBankAdapter
BankAdapter --* SberbankAPI

@enduml
```
