rectangle "Events" {
 class EventStorage {
  events: list<Event>
  + getEvents(): list<Event>
                + addEvent(e: Event) : bool
  + findEventById(id: int): Event

  + addEvent(e: Event): void
  + createMatch(eventId: int, match: Match): bool
  + reserveSeat(matchId: int, seatId: int): bool
  + updateEvent(id: int, newEvent: Event): bool
  + updateMatch(id: int, newMatch: Match): bool
 }
 class Event {
  id: int
  name: string
  matches: list<Match>
  eventTime: Tuple<int, int>
  + findMatchById(matchId: int): Match
  + addMatch(match: Match): bool
                + getId() : int
                + setId() : int
                + lastId() : int

 class Match {
  id: int
  name: string
                description: string
  priceCoefficient: double
  stadiumMap: StadiumMap

  + setSeatStatus(seatId: int, status: SeatStatus): bool
                + setSeatPrice(seatId: int, status: SeatStatus): bool
                + getMap() : StadiumMap
                + getId() : int
 }

 class StadiumMap {
  name: string
  layoutImage: string
  seats: list<Seat>
  turnstiles: list<int>
                + clone() : StadiumMap
                + setSeatStatus(status: SeatStatus, seatId: int) : bool
                + setSeatPrice(price: double, seatId: int) : bool
                + getImage(layoutImage: string) : string
 }

 class Seat {
  id: int
  sector: int
  row: int
  column: int
  price: double
                SeatStatus: status
                +getId() : int
                +setStatus(newStatus: SeatStatus) : bool
                +changePrice(price: double) : bool
                +getStatus() : SeatStatus
 }

 enum SeatStatus {
  Free
  Booked
  Sold
  Blocked
 }
}

rectangle "Passes" {
 class PassStorage {
  passes: list<Pass>
  + createPass(pass: Pass): void
  + checkRefundEligibility(passId: int): bool
  + refund(passId: int): bool
  + getPassById(passId: int): Pass
 }

 abstract class Pass {
  id: int
  seatId: int
  price: double
  email: string
  isRefunded: bool
  paymentMethod: PaymentType
  paymentCredentials: string
  + checkAccess(matchId: int): bool
  + markUsed(matchId: int): bool
  + checkRefundEligibility(): bool
 }

 class Ticket extends Pass {
  matchId: int
  used: bool
 }

 class Subscription extends Pass {
  matchIds: set<int>
  usedMatches: set<int>
  startTimestamp: int
  endTimestamp: int
 }


 enum PaymentType {
  Card
  Cash
 }
}
